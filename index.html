const express = require('express'); const cors = require('cors'); const fetch = require('node-fetch'); // O axios si prefieres const app = express(); const PORT = process.env.PORT || 3000; // Permitir peticiones desde tu frontend en Netlify app.use(cors()); app.use(express.json()); // API Key de JustAnotherPanel (o pásala por variable de entorno) const JAP_API_KEY = process.env.JAP_API_KEY || 'TU_API_KEY_DE_JAP_AQUI'; const JAP_API_URL = 'https://justanotherpanel.com/api/v2'; app.get('/', (req, res) => { res.send('Backend del Panel SMM está activo.'); }); // Ruta POST que recibe la orden desde index.html app.post('/api/order', async (req, res) => { const { service, link, quantity } = req.body; try { const response = await fetch(JAP_API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ key: JAP_API_KEY, action: 'add', service: service, link: link, quantity: quantity }) }); const data = await response.json(); res.json(data); } catch (error) { console.error('Error enviando orden a JAP:', error); res.status(500).json({ error: 'Error interno al procesar la orden con el proveedor.' }); } }); app.listen(PORT, () => { console.log(`Servidor corriendo en el puerto ${PORT}`); });
